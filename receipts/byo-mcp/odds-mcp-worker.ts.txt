import { type OddsQuestion, type OddsRequest, type OddsTransport, runOdds } from '/Users/jcoeyman/cloudflare/odds/src/client.ts';
import { ODDS_MODELS } from '/Users/jcoeyman/cloudflare/odds/src/models.ts';

type Env = { readonly ODDS_MCP_TOKEN?: string; readonly CF_ACCOUNT_ID: string; readonly CF_AI_TOKEN: string };

type JsonRpcRequest = { readonly jsonrpc: '2.0'; readonly id?: string | number | null; readonly method: string; readonly params?: Record<string, unknown> };

const PROTOCOL_VERSION = '2025-06-18';

const QUESTION_SCHEMA = {
  type: 'object',
  properties: {
    type: { type: 'string', enum: ['bool', 'choice', 'score'] },
    instructions: { type: 'string' },
    criteria: { description: 'choice: {label: description}. score: ordered list of 2 to 10 levels. bool: optional {true, false}.' },
  },
  required: ['type', 'instructions'],
};

const TOOLS = [
  {
    name: 'ask_odds',
    description: 'Ask Cloudflare Clef typed questions about a state and get calibrated probabilities back, not prose. Up to 64 questions per call.',
    inputSchema: {
      type: 'object',
      properties: {
        state: { description: 'The thing to judge: text or any JSON.' },
        questions: { type: 'object', additionalProperties: QUESTION_SCHEMA, description: 'Map of question id to question.' },
        model: { type: 'string', enum: ODDS_MODELS.map((model) => model.id) },
      },
      required: ['state', 'questions'],
    },
  },
  { name: 'odds_models', description: 'List the Clef models odds can ask.', inputSchema: { type: 'object', properties: {} } },
];

const workersAiTransport = (env: Env): OddsTransport => async (model, payload, signal) => {
  const response = await fetch(`https://api.cloudflare.com/client/v4/accounts/${env.CF_ACCOUNT_ID}/ai/run/${model.workersAiId}`, {
    method: 'POST',
    headers: { authorization: `Bearer ${env.CF_AI_TOKEN}`, 'content-type': 'application/json' },
    body: JSON.stringify(payload),
    signal,
  });
  return { status: response.status, body: await response.json().catch(() => ({ success: false, errors: [{ message: 'non-JSON reply' }] })) };
};

const toolResult = (value: unknown, isError = false) => ({
  content: [{ type: 'text', text: typeof value === 'string' ? value : JSON.stringify(value) }],
  ...(typeof value === 'object' && value !== null ? { structuredContent: value } : {}),
  isError,
});

const callTool = async (env: Env, name: string, args: Record<string, unknown>) => {
  if (name === 'odds_models') return toolResult({ models: ODDS_MODELS });
  if (name !== 'ask_odds') return toolResult(`unknown tool ${name}`, true);
  const request: OddsRequest = { state: args.state as OddsRequest['state'], questions: (args.questions ?? {}) as Record<string, OddsQuestion> };
  const result = await runOdds(workersAiTransport(env), typeof args.model === 'string' ? args.model : 'clef', request);
  return toolResult(result, !result.ok);
};

const answer = (id: JsonRpcRequest['id'], result: unknown) => ({ jsonrpc: '2.0', id: id ?? null, result });
const failure = (id: JsonRpcRequest['id'], code: number, message: string) => ({ jsonrpc: '2.0', id: id ?? null, error: { code, message } });

const handle = async (env: Env, message: JsonRpcRequest) => {
  if (message.method === 'initialize') {
    const asked = message.params?.protocolVersion;
    return answer(message.id, { protocolVersion: typeof asked === 'string' ? asked : PROTOCOL_VERSION, capabilities: { tools: { listChanged: false } }, serverInfo: { name: 'odds', version: '0.0.1' } });
  }
  if (message.method === 'tools/list') return answer(message.id, { tools: TOOLS });
  if (message.method === 'tools/call') return answer(message.id, await callTool(env, String(message.params?.name ?? ''), (message.params?.arguments ?? {}) as Record<string, unknown>));
  if (message.method === 'ping') return answer(message.id, {});
  return failure(message.id, -32601, `method not found: ${message.method}`);
};

const authorized = (request: Request, env: Env) => (env.ODDS_MCP_TOKEN ?? '') !== '' && request.headers.get('authorization') === `Bearer ${env.ODDS_MCP_TOKEN}`;

const isRpc = (item: unknown): item is JsonRpcRequest => typeof item === 'object' && item !== null && typeof (item as JsonRpcRequest).method === 'string';

export default {
  async fetch(request: Request, env: Env): Promise<Response> {
    if (new URL(request.url).pathname !== '/mcp') return new Response('odds MCP lives at /mcp', { status: 404 });
    if (!authorized(request, env)) return Response.json(failure(null, -32001, 'unauthorized'), { status: 401 });
    if (request.method === 'DELETE') return new Response(null, { status: 204 });
    if (request.method !== 'POST') return new Response(null, { status: 405, headers: { allow: 'POST' } });
    const body: unknown = await request.json().catch(() => null);
    const messages = (Array.isArray(body) ? body : [body]).filter(isRpc);
    if (messages.length === 0) return Response.json(failure(null, -32700, 'parse error'), { status: 400 });
    const replies = [];
    for (const message of messages) if (message.id !== undefined) replies.push(await handle(env, message));
    if (replies.length === 0) return new Response(null, { status: 202 });
    return Response.json(Array.isArray(body) ? replies : replies[0]);
  },
};
